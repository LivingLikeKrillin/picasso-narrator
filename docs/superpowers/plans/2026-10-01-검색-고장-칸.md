```python
def _names(value):
    """글자 목록이면 베껴 담고, 아니면 `None` — 모르는 모양을 추려 담으면 걸린 것이 빠져 온전해 보인다."""
    return list(value) if isinstance(value, list) and all(isinstance(x, str) for x in value) else None
```
